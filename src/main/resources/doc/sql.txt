评论表
CREATE TABLE `comments` (
  `comment_id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '评论唯一标识',
  `customer_id` BIGINT NOT NULL COMMENT '关联顾客表',
  `content` TEXT NOT NULL COMMENT '评论内容',
  `domain_id` BIGINT NOT NULL COMMENT '评论所属领域：0=外卖(waimai)，1=电商(shop)，2=酒店(hotel)',
  `publish_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '评论发布时间',
  `product_id` BIGINT NOT NULL COMMENT '关联商品表',
  `merchant_id` BIGINT NOT NULL COMMENT '关联商家表',
  `status` INTEGER NOT NULL DEFAULT 0 COMMENT '评论当前状态：0=待推理(pending)，1=推理中(inferring)，2=已推理(inferred)，3=审核中(reviewing)，4=已修正(corrected)，5=已拒绝(rejected)',
  `final_sentiment` INTEGER COMMENT '最终情感结果：1=好评(positive)，0=差评(negative)',
  `is_retrained` BOOLEAN NOT NULL DEFAULT FALSE COMMENT '是否被重训过',
  PRIMARY KEY (`comment_id`),
  INDEX `idx_customer_id` (`customer_id`),
  INDEX `idx_domain_id` (`domain_id`),
  INDEX `idx_product_id` (`product_id`),
  INDEX `idx_merchant_id` (`merchant_id`),
  INDEX `idx_publish_time` (`publish_time`),
  INDEX `idx_status` (`status`),
  INDEX `idx_final_sentiment` (`final_sentiment`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='商家情感分析-评论主表';

推理记录表
CREATE TABLE `inference_records` (
  `inference_id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '推理记录ID',
  `inference_task_id` BIGINT NOT NULL COMMENT '推理任务ID',
  `comment_id` BIGINT NOT NULL COMMENT '关联评论表',
  `model_id` BIGINT NOT NULL COMMENT '关联模型表',
  `domain_id` BIGINT  NOT NULL COMMENT '评论所属领域',
  `positive_prob` DECIMAL(5,4) NOT NULL COMMENT '模型推理的正面概率',
  `negative_prob` DECIMAL(5,4) NOT NULL COMMENT '模型推理的负面概率',
  `model_sentiment` INTEGER NOT NULL COMMENT '模型判断结果',

  `confidence` DECIMAL(4,3) NOT NULL COMMENT '推理置信度',

  INDEX `idx_model_id` (`model_id`),

模型表
CREATE TABLE models (
    model_id      BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '模型ID',
    domain_id     BIGINT NOT NULL COMMENT '领域id',
    model_version VARCHAR(20) NOT NULL COMMENT '模型版本号',
    source        ENUM('train', 'upload') NOT NULL COMMENT '来源',
    inferred_num BIGINT NOT NULL DEFAULT 0 COMMENT '总推理数',
    corrected_num  BIGINT NOT NULL DEFAULT 0 COMMENT '被修正数(错误数)',
    deleted      Integer default 0 not null COMMENT '1表示删除，0表示没被删除';
    base_model_id BIGINT DEFAULT NULL COMMENT '模型对应的，基础模型id'
    -- 虚拟生成的准确率字段
    -- 逻辑：如果总数为0则为0，否则计算 (1 - 修正/总数) * 100
    accuracy      DECIMAL(5, 2) GENERATED ALWAYS AS (
        CASE
            WHEN inferred_num = 0 THEN 0
            ELSE (1 - (corrected_num / inferred_num)) * 100
        END
    ) VIRTUAL COMMENT '最终准确率%',

    train_task_id BIGINT DEFAULT NULL,
    description   TEXT,
    created_at    DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_domain_id (domain_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='模型表';

训练任务表
CREATE TABLE `train_tasks` (
    -- 1. 基础信息
    `id` BIGINT AUTO_INCREMENT PRIMARY KEY COMMENT '训练任务id',
    `creator_id` BIGINT DEFAULT 0 NOT NULL COMMENT '发起人id',
    `domain_id` BIGINT NOT NULL COMMENT '领域id',
    `domain_name` VARCHAR(100) NOT NULL COMMENT '领域中文名字',
    `status` INTEGER NOT NULL DEFAULT 0 COMMENT '训练状态 (0:待处理, 1:处理中, 2:成功完成, 3:失败完成)',
    `status_msg` TEXT NULL COMMENT '训练状态信息，用于存储错误记录',

    -- 2. 时间与统计
    `start_time` DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '训练开始时间',
    `end_time` DATETIME DEFAULT NULL COMMENT '训练结束时间',
    `duration` BIGINT DEFAULT NULL COMMENT '训练持续时间(秒)',
    `total_data` BIGINT DEFAULT 0 COMMENT '训练总评论数',

    -- 3. 训练参数配置
    `correctedNum` BIGINT DEFAULT 0 COMMENT '来源于评论审核的数据数量',
    `uploadNum` BIGINT DEFAULT 0 COMMENT '来源于手动上传的数据数量',
    `originalNum` BIGINT DEFAULT 0 COMMENT '来源于初始数据集的数据数量',
    `lora_r` INTEGER  DEFAULT 8 COMMENT 'LORA_R',
    `lora_alpha` INTEGER  DEFAULT 16 COMMENT 'lora_alpha',
    `epochs` INTEGER  DEFAULT 5 COMMENT '训练轮数',
    `batch_size` INTEGER  DEFAULT 64 COMMENT '训练批次大小',
    `learning_rate` DECIMAL(10, 8) DEFAULT 0.00001000 COMMENT '学习率(1e-5)',
    `random_seed` INTEGER  DEFAULT 42 COMMENT '随机种子',
    `lora_modules` JSON COMMENT 'Lora模块列表, eg: ["query", "key", "value", "dense"]',
    `train_split_ratio` DECIMAL(3, 2) DEFAULT 0.80 COMMENT '训练集比例, eg: 0.80',
    `if_over_train` INTEGER default 0 COMMENT '是否重新训练，是=1 否=0',

    -- 4. 训练产出与指标
    `model_id` BIGINT  DEFAULT NULL COMMENT '训练得到的模型id',
    `model_version` VARCHAR(20) NULL COMMENT '模型版本号(两位小数, 如1.12)',
    `accuracy` DECIMAL(5, 4) DEFAULT NULL COMMENT '总体准确率',
    `precision_rate` DECIMAL(5, 4) DEFAULT NULL COMMENT '精确率',
    `recall_rate` DECIMAL(5, 4) DEFAULT NULL COMMENT '召回率',
    `f1_score` DECIMAL(5, 4) DEFAULT NULL COMMENT 'f1分数',

    -- 5. 可视化曲线数据 (JSON存储列表)
    `train_loss_list` JSON COMMENT '训练集loss变化趋势数组',
    `train_acc_list` JSON COMMENT '训练集准确率变化趋势数组',
    `val_loss_list` JSON COMMENT '验证集loss变化趋势数组',
    `val_acc_list` JSON COMMENT '验证集准确率变化趋势数组',

    -- 6. 通用审计字段
    `create_time` DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '记录创建时间',
    `update_time` DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '记录更新时间',

    -- 索引优化
    INDEX `idx_creator` (`creator_id`),
    INDEX `idx_domain` (`domain_id`),
    INDEX `idx_status` (`status`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='训练任务管理表';




训练参数表

    `para_id` BIGINT AUTO_INCREMENT PRIMARY KEY COMMENT '参数配置Id',
    `domain_id` BIGINT NOT NULL COMMENT '领域id',

    -- 训练核心参数
    `lora_r` INT NOT NULL DEFAULT 8 COMMENT 'LORA_R',

    `random_seed` INT NOT NULL DEFAULT 42 COMMENT '随机种子',

    -- 扩展参数
    `lora_modules` JSON COMMENT 'Lora模块列表, 可能的值: ["query", "key", "value", "dense"]',
    `train_split_ratio` DECIMAL(3, 2) NOT NULL DEFAULT 0.80 COMMENT '训练集比例, eg: 0.80',

    -- 审计字段
    `create_time` DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    `update_time` DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',

    -- 索引
    INDEX `idx_domain` (`domain_id`),
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='训练参数配置模板表';


训练数据表
CREATE TABLE `train_data` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '训练数据id (主键)',
    `domain_id` BIGINT NOT NULL COMMENT '领域id',
    `content` TEXT NOT NULL COMMENT '训练数据内容',
    `label` INT NOT NULL COMMENT '训练数据标签 0差1好',
    `source` ENUM('corrected', 'upload', 'original') NOT NULL COMMENT '训练数据来源 (corrected/upload/original)',
    `train_count` INT UNSIGNED NOT NULL DEFAULT 0 COMMENT '数据被训练次数',
    `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '数据创建时间',
    PRIMARY KEY (`id`),
    -- 如果需要按来源筛选，可以增加索引
    INDEX `idx_source` (`source`),
    -- 如果需要按创建时间排序，可以增加索引
    INDEX `idx_created_at` (`created_at`),
    -- 按领域筛选/统计训练数据
    INDEX `idx_domain` (`domain_id`),
    -- 训练面板按领域+来源聚合时使用
    INDEX `idx_domain_source` (`domain_id`, `source`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='训练数据表';
